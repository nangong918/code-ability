# 数据结构与算法


#### 遍历二叉树的log n是怎么算出来的？一定是log n吗？

二叉树搜素的原理：
- 从根节点开始比较
- 目标值更小 → 走左子树；目标值更大 → 走右子树
- 每往下走一层，搜索范围直接砍掉一半
- 最多走树的高度 h 就一定能找到 / 确定不存在

完全二叉树
* 节点总数：n
* 树的高度：h（根节点算第 1 层）

深度为 h 的满二叉树，最大节点数公式：

`n = 2^h - 1`

目标是求h：`h = log(n + 1) ≈ log(n)`

如果是`斜树的二叉搜索树`则退化成n


#### 讲讲递归的时间复杂度以及递归和二叉树的关系

二叉树就是递归的2分遍历结构，真实的应该是n叉树，所以递归的查找也是找到高度h，
也就是说递归的时空复杂度是log以m为底的(n + 1)， 所以搜索复杂度也是log(n)

最坏情况也是斜状树，斜状树的搜索复杂度退化到n

###### 2.1 冒泡排序

```cpp
#include <iostream>
#include <vector>
using namespace std;

// 冒泡排序：稳定、原地排序
void bubbleSort(vector<int>& arr) {
    int n = arr.size();
    bool flag; // 优化：标记是否已有序
    for (int i = 0; i < n - 1; i++) {
        flag = false;
        for (int j = 0; j < n - 1 - i; j++) {
            if (arr[j] > arr[j + 1]) {
                swap(arr[j], arr[j + 1]);
                flag = true;
            }
        }
        if (!flag) break; // 已有序，提前退出
    }
}

// 测试
int main() {
    vector<int> arr = {5, 2, 9, 1, 5, 6};
    bubbleSort(arr);
    for (int num : arr) cout << num << " ";
    return 0;
}
```

###### 2.2 选择排序

```cpp
#include <iostream>
#include <vector>
using namespace std;

// 选择排序：不稳定、原地排序
void selectSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n - 1; i++) {
        int minIdx = i;
        // 找到未排序区间最小值下标
        for (int j = i + 1; j < n; j++) {
            if (arr[j] < arr[minIdx]) minIdx = j;
        }
        swap(arr[i], arr[minIdx]);
    }
}

// 测试
int main() {
    vector<int> arr = {5, 2, 9, 1, 5, 6};
    selectSort(arr);
    for (int num : arr) cout << num << " ";
    return 0;
}
```

###### 2.3 插入排序

```cpp
#include <iostream>
#include <vector>
using namespace std;

// 插入排序：稳定、原地排序，有序数组效率极高
void insertSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 1; i < n; i++) {
        int temp = arr[i];
        int j = i - 1;
        // 后移大于temp的元素
        while (j >= 0 && arr[j] > temp) {
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = temp;
    }
}

// 测试
int main() {
    vector<int> arr = {5, 2, 9, 1, 5, 6};
    insertSort(arr);
    for (int num : arr) cout << num << " ";
    return 0;
}
```

###### 2.4 希尔排序

```cpp
#include <iostream>
#include <vector>
using namespace std;

// 希尔排序：插入排序优化版、不稳定、原地排序
void shellSort(vector<int>& arr) {
    int n = arr.size();
    // 增量序列：折半递减
    for (int gap = n / 2; gap > 0; gap /= 2) {
        // 分组插入排序
        for (int i = gap; i < n; i++) {
            int temp = arr[i];
            int j;
            for (j = i; j >= gap && arr[j - gap] > temp; j -= gap) {
                arr[j] = arr[j - gap];
            }
            arr[j] = temp;
        }
    }
}

// 测试
int main() {
    vector<int> arr = {5, 2, 9, 1, 5, 6};
    shellSort(arr);
    for (int num : arr) cout << num << " ";
    return 0;
}
```

###### 2.5 快速排序

```cpp
#include <iostream>
#include <vector>
using namespace std;

// 快速排序：不稳定、递归排序，平均效率最高
void quickSort(vector<int>& arr, int left, int right) {
    if (left >= right) return;
    int l = left, r = right;
    int pivot = arr[left]; // 基准值

    while (l < r) {
        // 从右找小于基准的数
        while (l < r && arr[r] >= pivot) r--;
        arr[l] = arr[r];
        // 从左找大于基准的数
        while (l < r && arr[l] <= pivot) l++;
        arr[r] = arr[l];
    }
    arr[l] = pivot;
    // 递归排序左右区间
    quickSort(arr, left, l - 1);
    quickSort(arr, l + 1, right);
}

// 重载统一调用
void quickSort(vector<int>& arr) {
    quickSort(arr, 0, arr.size() - 1);
}

// 测试
int main() {
    vector<int> arr = {5, 2, 9, 1, 5, 6};
    quickSort(arr);
    for (int num : arr) cout << num << " ";
    return 0;
}
```

###### 2.6 归并排序

```cpp
#include <iostream>
#include <vector>
using namespace std;

// 归并排序：稳定、分治思想、非原地排序
void merge(vector<int>& arr, int left, int mid, int right) {
    vector<int> temp(right - left + 1);
    int i = left, j = mid + 1, k = 0;
    // 合并两个有序区间
    while (i <= mid && j <= right) {
        if (arr[i] <= arr[j]) temp[k++] = arr[i++];
        else temp[k++] = arr[j++];
    }
    while (i <= mid) temp[k++] = arr[i++];
    while (j <= right) temp[k++] = arr[j++];
    // 回填原数组
    for (int p = 0; p < temp.size(); p++) {
        arr[left + p] = temp[p];
    }
}

void mergeSort(vector<int>& arr, int left, int right) {
    if (left >= right) return;
    int mid = (left + right) / 2;
    mergeSort(arr, left, mid);
    mergeSort(arr, mid + 1, right);
    merge(arr, left, mid, right);
}

void mergeSort(vector<int>& arr) {
    mergeSort(arr, 0, arr.size() - 1);
}

// 测试
int main() {
    vector<int> arr = {5, 2, 9, 1, 5, 6};
    mergeSort(arr);
    for (int num : arr) cout << num << " ";
    return 0;
}
```

###### 2.7 堆排序

```cpp
#include <iostream>
#include <vector>
using namespace std;

// 堆排序：不稳定、原地排序，基于大顶堆
// 调整堆
void heapify(vector<int>& arr, int n, int i) {
    int maxIdx = i;
    int l = 2 * i + 1;
    int r = 2 * i + 2;
    if (l < n && arr[l] > arr[maxIdx]) maxIdx = l;
    if (r < n && arr[r] > arr[maxIdx]) maxIdx = r;
    if (maxIdx != i) {
        swap(arr[i], arr[maxIdx]);
        heapify(arr, n, maxIdx);
    }
}

void heapSort(vector<int>& arr) {
    int n = arr.size();
    // 构建大顶堆
    for (int i = n / 2 - 1; i >= 0; i--) {
        heapify(arr, n, i);
    }
    // 逐步取出堆顶元素
    for (int i = n - 1; i > 0; i--) {
        swap(arr[0], arr[i]);
        heapify(arr, i, 0);
    }
}

// 测试
int main() {
    vector<int> arr = {5, 2, 9, 1, 5, 6};
    heapSort(arr);
    for (int num : arr) cout << num << " ";
    return 0;
}
```

###### 2.8 计数排序

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

// 计数排序：稳定、适合数值范围小的整数数组
void countSort(vector<int>& arr) {
    int n = arr.size();
    if (n == 0) return;
    int maxVal = *max_element(arr.begin(), arr.end());
    int minVal = *min_element(arr.begin(), arr.end());
    int range = maxVal - minVal + 1;

    vector<int> count(range, 0);
    // 统计次数
    for (int num : arr) count[num - minVal]++;
    // 回填数组
    int idx = 0;
    for (int i = 0; i < range; i++) {
        while (count[i]--) {
            arr[idx++] = i + minVal;
        }
    }
}

// 测试
int main() {
    vector<int> arr = {5, 2, 9, 1, 5, 6};
    countSort(arr);
    for (int num : arr) cout << num << " ";
    return 0;
}
```

##### 三、 搜索算法 C++ 完整实现代码

###### 3.1 顺序查找（线性查找）

```cpp
#include <iostream>
#include <vector>
using namespace std;

// 顺序查找：无序数组通用，返回下标，未找到返回-1
int linearSearch(vector<int>& arr, int target) {
    for (int i = 0; i < arr.size(); i++) {
        if (arr[i] == target) return i;
    }
    return -1;
}

// 测试
int main() {
    vector<int> arr = {5, 2, 9, 1, 5, 6};
    int pos = linearSearch(arr, 9);
    cout << "目标下标：" << pos << endl;
    return 0;
}
```

###### 3.2 二分查找（迭代版，最优）

```cpp
#include <iostream>
#include <vector>
using namespace std;

// 二分查找：仅适用于有序数组，迭代版空间O(1)
int binarySearch(vector<int>& arr, int target) {
    int left = 0, right = arr.size() - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2; // 避免溢出
        if (arr[mid] == target) return mid;
        else if (arr[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}

// 测试
int main() {
    vector<int> arr = {1, 2, 5, 5, 6, 9}; // 有序数组
    int pos = binarySearch(arr, 5);
    cout << "目标下标：" << pos << endl;
    return 0;
}
```

###### 3.3 二分查找（递归版）

```cpp
#include <iostream>
#include <vector>
using namespace std;

int binarySearchRecursion(vector<int>& arr, int left, int right, int target) {
    if (left > right) return -1;
    int mid = left + (right - left) / 2;
    if (arr[mid] == target) return mid;
    else if (arr[mid] < target) {
        return binarySearchRecursion(arr, mid + 1, right, target);
    } else {
        return binarySearchRecursion(arr, left, mid - 1, target);
    }
}

// 测试
int main() {
    vector<int> arr = {1, 2, 5, 5, 6, 9};
    int pos = binarySearchRecursion(arr, 0, arr.size() - 1, 5);
    cout << "目标下标：" << pos << endl;
    return 0;
}
```

###### 3.4 哈希查找（unordered_map）

```cpp
#include <iostream>
#include <vector>
#include <unordered_map>
using namespace std;

// 哈希查找：平均O(1)查找效率
int hashSearch(vector<int>& arr, int target) {
    unordered_map<int, int> hashMap;
    // 构建哈希表
    for (int i = 0; i < arr.size(); i++) {
        hashMap[arr[i]] = i;
    }
    // 查找
    if (hashMap.find(target) != hashMap.end()) {
        return hashMap[target];
    }
    return -1;
}

// 测试
int main() {
    vector<int> arr = {5, 2, 9, 1, 5, 6};
    int pos = hashSearch(arr, 9);
    cout << "目标下标：" << pos << endl;
    return 0;
}
```

## 3.5 二叉搜索树查找

```cpp
#include <iostream>
using namespace std;

// 定义二叉搜索树节点
struct TreeNode {
    int val;
    TreeNode* left;
    TreeNode* right;
    TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
};

// 二叉搜索树查找
TreeNode* bstSearch(TreeNode* root, int target) {
    if (!root || root->val == target) return root;
    // 小于根节点查左子树，大于查右子树
    if (target < root->val) {
        return bstSearch(root->left, target);
    } else {
        return bstSearch(root->right, target);
    }
}

// 测试
int main() {
    // 构建简单BST
    TreeNode* root = new TreeNode(5);
    root->left = new TreeNode(3);
    root->right = new TreeNode(7);
    root->left->left = new TreeNode(2);

    TreeNode* res = bstSearch(root, 3);
    if (res) cout << "查找成功，值为：" << res->val << endl;
    else cout << "查找失败" << endl;
    return 0;
}
```
